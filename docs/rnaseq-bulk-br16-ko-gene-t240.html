dge_list <- foreach(use_subset = subset_list) %do%{
  use_se <- se[,use_subset]
  group_levels <- outer(levels(use_se$condition), levels(use_se$ko), paste, sep="_") %>% 
    as.vector
  use_se$group <- paste(use_se$condition, use_se$ko, sep = '_') %>% 
    factor(levels = group_levels)

  # Generate DESeq for group
  dds <- DESeqDataSet(use_se, design = ~ group)
  dds <- DESeq(dds)
   # resultsNames(dds)
  # Contrast: a character vector with exactly three elements: the name of a factor in the design formula, the name of the numerator level for the fold change, and the name of the denominator level for the fold change (simplest case)
  res_chip_in_wt <- results(dds, contrast = c("group", "Chip_wt", "Control_wt")) %>% 
    data.frame %>% 
    rownames_to_column('gene_name') %>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
  res_chip_in_ko <- results(dds, contrast = c("group", "Chip_ko", "Control_ko")) %>% 
    data.frame %>% 
    rownames_to_column('gene_name') %>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
  res_ko_in_control <- results(dds, contrast = c("group", "Control_ko", "Control_wt")) %>% 
    data.frame %>% 
    rownames_to_column('gene_name') %>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
  res_ko_in_chip <- results(dds, contrast = c("group", "Chip_ko", "Chip_wt")) %>% 
    data.frame %>% 
    rownames_to_column('gene_name') %>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
   
  
  # Generate DESeq for interaction
  dds <- DESeqDataSet(use_se, design = ~ condition + ko + condition:ko)
  dds <- DESeq(dds)
  # resultsNames(dds)
  # res_chip_wt_i <- results(dds, contrast = c("condition", "Control", "Chip"))
  # res_ko_in_control_i <- results(dds, contrast = c("ko", "wt", "ko"))
  res_i <- results(dds, name ="conditionChip.koko")  %>% 
    data.frame %>% 
    rownames_to_column('gene_name') %>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
  # res_chip_wt== res_chip_wt_i
  # res_ko_in_control == res_ko_in_control_i
  
  # Interpretation of log2FC of interaction:
  # the log2FC of interaction (conditionChip.koko) represents the 
  # additional effect of ko in Chip condition compared to Control condition. 
  # log2FC(i) = log2FC(Chip_ko-over-Chip_wt) - log2FC(Control_ko-over-Control_wt)
  
  # Combine results in a single data.frame
  res_comb <- full_join(
    res_chip_in_wt,
    res_chip_in_ko,
    by = c('gene_name', 'description'), 
    suffix = c('_wt', '_ko')
  ) %>% 
    left_join(
      res_i %>% 
        dplyr::rename(
          log2FoldChange_int = log2FoldChange, 
          pvalue_int = pvalue, 
          padj_int = padj) %>% 
        dplyr::select(gene_name, ends_with('int'))
      ) %>% 
    dplyr::select(-description)%>% 
    left_join(
      gene_desc, by = 'gene_name'
    )
    
  
  # Return results
  res <- list(
    chip_in_wt = res_chip_in_wt,
    chip_in_ko = res_chip_in_ko,
    ko_in_control = res_ko_in_control,
    ko_in_chip = res_ko_in_chip,
    interaction = res_i,
    combined = res_comb
  )
  
  return(res)
}
